```bash
#!/usr/bin/env bash
set -u

HOSTS=(
  k8s-node-1
  k8s-node-2
  k8s-node-3
)

KEY="$HOME/.ssh/my_private_key"
KNOWN_HOSTS="$HOME/.ssh/my_known_hosts"
REMOTE_USER="user"
OUTPUT="./kesl_status_all.txt"

# Начать новый файл результатов
: > "$OUTPUT"

for host in "${HOSTS[@]}"; do
  {
    echo "========================================"
    echo "Хост: $host"
    echo "Дата: $(date '+%Y-%m-%d %H:%M:%S')"
    echo "========================================"

    if ssh \
      -i "$KEY" \
      -o UserKnownHostsFile="$KNOWN_HOSTS" \
      -o StrictHostKeyChecking=yes \
      -o BatchMode=yes \
      -T \
      "${REMOTE_USER}@${host}" \
      "sudo -n systemctl status kesl --no-pager" 2>&1
    then
      echo "Результат: SSH-команда завершилась успешно"
    else
      rc=$?
      echo "Результат: ошибка, код завершения $rc"
    fi

    echo
  } >> "$OUTPUT"
done

echo "Все результаты сохранены в $OUTPUT"
```
