#!/usr/bin/env bash
set -uo pipefail

HOSTS=(
  k8s-node-1
  k8s-node-2
  k8s-node-3
)

KEY="$HOME/.ssh/my_private_key"
KNOWN_HOSTS="$HOME/.ssh/my_known_hosts"
REMOTE_USER="user"

for host in "${HOSTS[@]}"; do
  output="./${host}_kesl_status.txt"
  tmp="${output}.tmp"

  echo "Проверка ${host}..."

  if ssh \
    -i "$KEY" \
    -o UserKnownHostsFile="$KNOWN_HOSTS" \
    -o StrictHostKeyChecking=yes \
    -o BatchMode=yes \
    -T \
    "${REMOTE_USER}@${host}" \
    "sudo -n systemctl status kesl --no-pager" \
    > "$tmp" 2>&1
  then
    mv -- "$tmp" "$output"
    echo "Результат сохранён: $output"
  else
    rc=$?
    mv -- "$tmp" "$output"
    echo "Команда завершилась с кодом $rc на ${host}. Вывод: $output" >&2
  fi
done
