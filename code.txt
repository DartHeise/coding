#!/usr/bin/env bash
set -u

HOSTS=(
  k8s-node-1
  k8s-node-2
  k8s-node-3
)

KEY="$HOME/.ssh/my_private_key"
KNOWN_HOSTS="$HOME/.ssh/my_known_hosts"
REMOTE_USER="user"

for host in "${HOSTS[@]}"; do
  ssh \
    -i "$KEY" \
    -o UserKnownHostsFile="$KNOWN_HOSTS" \
    -o StrictHostKeyChecking=yes \
    -o BatchMode=yes \
    -T \
    "${REMOTE_USER}@${host}" \
    "sudo -n systemctl stop kesl" \
    >/dev/null 2>&1
done
